const express = require('express')
const formidable = require('formidable').default
const fs = require('fs')
const luxon = require('luxon')
const path = require('path')
const { randomUUID } = require('crypto') 

const DateTime = luxon.DateTime

module.exports = ({ auth, func, db, valid }) => {
    const router = express.Router()

    const moduleName = 'docs'
    const folderOptions = [
        { value: 'general', label: 'General Documents' },
        { value: 'policies', label: 'Policies' },
        { value: 'guides', label: 'Guides' },
        { value: 'archives', label: 'Archives' }
    ]

    function deleteFile(filePath, cb = () => {}) {
        if (!filePath) {
            cb()
        } else {
            fs.unlink(filePath, err => {
                if (err && err.code !== 'ENOENT') {
                    console.error('Error deleting file:', err)
                }

                cb()
            })
        }
    }

    function formatFileSize(fileSize) {
        const size = Number(fileSize) || 0

        if (size < 1024) {
            return `${size} B`
        }

        if (size < 1024 * 1024) {
            return `${(size / 1024).toFixed(1)} KB`
        }

        if (size < 1024 * 1024 * 1024) {
            return `${(size / (1024 * 1024)).toFixed(1)} MB`
        }

        return `${(size / (1024 * 1024 * 1024)).toFixed(1)} GB`
    }

    function getFeedbackFromStatus(status) {
        switch (status) {
            case 'uploaded':
                return { type: 'success', message: 'Document uploaded and saved successfully.' }
            case 'deleted':
                return { type: 'success', message: 'Document deleted successfully.' }
            case 'upload-error':
                return { type: 'danger', message: 'The document could not be uploaded.' }
            case 'delete-error':
                return { type: 'danger', message: 'The document could not be deleted.' }
            case 'delete-mismatch':
                return { type: 'danger', message: 'Delete confirmation did not match the document name.' }
            case 'forbidden':
                return { type: 'danger', message: 'You do not have permission to perform that action.' }
            default:
                return null
        }
    }

    /**
     * Takes in the user perms object, if they are an admin or file manager it will return all permissions, otherwise it will just show the permissions that user has.
     * @param {Object} userPerms 
     * @returns {Array}
     */
    function getPermissionsList(userPerms) {
        const allPermissionOptions = global.permData.map(permission => permission.permissionName)

        if (func.hasPermissions(['ADMINISTRATOR', 'MANAGE_FILES'], userPerms, 'OR')) {
            return allPermissionOptions
        } else {
            return allPermissionOptions.filter(perm => userPerms[perm] === true)
        }
    }

    router.get('/', (req, res) => {
        func.getUserData(req.user, (success, userData) => {
            if (!success) {
                console.error('Error fetching user data:', userData)
                res.status(500).send('Unable to load page')
            } else {
                db.query('SELECT docs.*, users.username AS uploadedByUsername FROM docs LEFT JOIN users ON docs.uploadedBy = users.userID ORDER BY docs.targetFolder ASC, docs.uploadedAt DESC, docs.documentName ASC', (err, results) => {
                        if (err) {
                            console.error('Error fetching documents:', err)
                            res.status(500).send('Unable to load documents')
                        } else {
                            // Determines if they are admin or have universal file management permissions, it will override everything else.
                            const isGlobalFileManager = func.hasPermissions(['ADMINISTRATOR', 'MANAGE_FILES'], userData.perms, 'OR')

                            // Will calculate various fields, such as the folder it will sit in's label, delete or view the file and return an object to be used in the rendering.
                            const documents = (results || [])
                            .map(document => {
                                const uploadedAt = DateTime.fromSeconds(document.uploadedAt)
                                const folder = folderOptions.find(option => option.value === document.targetFolder)
                                const canDelete = isGlobalFileManager || func.hasPermissions([document.deletePermission], userData.perms, 'AND')
                                const canView = document.isPublic === 1 || canDelete

                                return {
                                    ...document,
                                    isPublic: document.isPublic === 1,
                                    uploadedAtDisplay: uploadedAt && uploadedAt.isValid ? uploadedAt.toFormat('dd LLL yyyy, HH:mm') : 'Unknown',
                                    fileSizeDisplay: formatFileSize(document.fileSize),
                                    folderLabel: folder.label,
                                    canDelete,
                                    canView,
                                    uploadedByDisplay: document.uploadedByUsername || document.uploadedBy || 'Unknown'
                                }
                            })
                            .filter(document => document.canView)


                            const folders = folderOptions.map(folder => ({
                                ...folder,
                                documents: documents.filter(document => document.targetFolder === folder.value)
                            }))

                            res.render('default', {
                                isAuthenticated: req.isAuthenticated(),
                                userData,
                                pagePath: `module-${moduleName}/home`,
                                pageTitle: 'Docs - MCMS',
                                feedback: getFeedbackFromStatus(req.query.status),
                                folders,
                                canUpload: userData.permsValue && userData.permsValue > 0
                            })
                        }
                    }
                )
            }
        })
    })

    router.route('/upload')
    .get(auth.isAuthorised, (req, res) => {
        func.getUserData(req.user, (success, userData) => {
            if (!success) {
                console.error('Error fetching user data:', userData)
                res.status(500).send('Unable to load page')
            } else {
                if (userData.permsValue === 0) {
                    res.redirect('/forbidden')
                } else {
                    res.render('default', {
                        isAuthenticated: req.isAuthenticated(),
                        userData,
                        pagePath: `module-${moduleName}/upload`,
                        pageTitle: 'Upload Document - MCMS',
                        permissionOptions: getPermissionsList(userData.perms),
                        folderOptions,
                        feedback: null,
                        formData: {}
                    })
                }
            }
        })
    })
    .post(auth.isAuthorised, (req, res) => {
        func.getUserData(req.user, (success, userData) => {
            if (!success) {
                console.error('Error fetching user data:', userData)
                res.status(500).send('Unable to load page')
            } else {
                if (userData.permsValue === 0) {
                    res.redirect('/forbidden')
                } else {
                    const uploadRootPath = path.resolve(__dirname, '..', 'uploads')
                    fs.mkdirSync(uploadRootPath, { recursive: true }) // Creates the folder if it dosen't already exist

                    const form = formidable({
                        uploadDir: uploadRootPath,
                        keepExtensions: true,
                        allowEmptyFiles: false,
                        minFileSize: 1,
                        maxFiles: 1,
                        maxFileSize: 25 * 1024 * 1024
                    })
                
                    form.parse(req, (err, fields, files) => {
                        if (err) {
                            console.error('Error parsing upload form:', err)
                                
                            res.status(400).render('default', {
                                isAuthenticated: req.isAuthenticated(),
                                userData,
                                pagePath: `module-${moduleName}/upload`,
                                pageTitle: 'Upload Document - MCMS',
                                permissionOptions: getPermissionsList(userData.perms),
                                folderOptions,
                                feedback: {
                                    type: 'danger',
                                    message: 'The upload could not be processed. Please check the file and try again.'
                                },
                                formData: {}
                            })
                        } else {
                            // Santise the inputs to ensure there are no leading or trailing spaces or characters that could cause any issues.
                            const documentName = valid.sanitiseInput(fields.documentName[0].trim())
                            const deletePermission = valid.sanitiseInput(fields.deletePermission[0].trim())
                            const targetFolder = valid.sanitiseInput(fields.targetFolder[0].trim())
                            const isPublic = fields.isPublic[0] === 'true'
                            const uploadedFile = files.documentFile[0]
                            
                            const formData = {
                                documentName,
                                deletePermission,
                                targetFolder,
                                isPublic
                            }
                            
                            /**
                             * If there was an upload error, it will try to delete the file that has already been saved, and then render the page with an error message.
                             * @param {string} message 
                             */
                            const uploadError = (message) => {
                                deleteFile(uploadedFile ? uploadedFile.filepath : null, () => {
                                    res.status(400).render('default', {
                                        isAuthenticated: req.isAuthenticated(),
                                        userData,
                                        pagePath: `module-${moduleName}/upload`,
                                        pageTitle: 'Upload Document - MCMS',
                                        permissionOptions: getPermissionsList(userData.perms),
                                        folderOptions: folderOptions.some(folder => folder.value === targetFolder),
                                        feedback: {
                                            type: 'danger',
                                            message
                                        },
                                        formData
                                    })
                                })
                            }
                        
                            // Performs server side checks on the files
                            if (!(documentName.length >= 3 && documentName.length <= 25)) {
                                uploadError('Document name must be between 3 and 255 characters long.')
                                return
                            }
                            
                            if (!getPermissionsList(userData.perms).includes(deletePermission)) {
                                uploadError('Please select a valid permission for deleting.')
                                return
                            }
                            
                            if (!folderOptions.some(folder => folder.value === targetFolder)) {
                                uploadError('Please select a valid destination folder.')
                                return
                            }
                            
                            if (!uploadedFile || !uploadedFile.originalFilename || !uploadedFile.filepath) {
                                uploadError('Please choose a file to upload.')
                                return
                            }
                            
                            const targetFolderPath = path.join(uploadRootPath, targetFolder)
                            const storedFilename = `${randomUUID()}${path.extname(uploadedFile.originalFilename || '').toLowerCase()}`
                            const finalFilePath = path.join(targetFolderPath, storedFilename)
                            const relativeFilePath = path.join(targetFolder, storedFilename).replace(/\\/g, '/')
                            const uploadedAt = Math.floor(DateTime.now().toSeconds())
                            
                            fs.mkdirSync(targetFolderPath, { recursive: true })

                            
                            fs.rename(uploadedFile.filepath, finalFilePath, err => {
                                if (err) {
                                    console.error('Error moving uploaded file:', err)
                                    deleteFile(uploadedFile.filepath)
                                    res.redirect('/docs?status=upload-error')

                                    return
                                } else {
                                    db.query('INSERT INTO docs (docID, documentName, originalFilename, storedFilename, filePath, mimeType, fileSize, deletePermission, targetFolder, isPublic, uploadedBy, uploadedAt) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)',
                                        [
                                            randomUUID(),
                                            documentName,
                                            uploadedFile.originalFilename,
                                            storedFilename,
                                            relativeFilePath,
                                            uploadedFile.mimetype || null,
                                            uploadedFile.size || 0,
                                            deletePermission,
                                            targetFolder,
                                            isPublic ? 1 : 0,
                                            req.user || null,
                                            uploadedAt
                                        ], (err) => {
                                            if (err) {
                                                console.error('Error inserting document record:', err)
                                                deleteFile(finalFilePath, () => {
                                                    res.redirect('/docs?status=upload-error')
                                                })
                                            } else {
                                                res.redirect('/docs?status=uploaded')
                                            }
                                        }
                                    )
                                }
                            })
                        }
                    })
                }
            }
        })
    })

    router.post('/delete/:docID', auth.isAuthorised, (req, res) => {
        func.getUserData(req.user, (success, userData) => {
            if (!success) {
                console.error('Error fetching user data:', userData)
                res.redirect('/docs?status=forbidden')
            } else {
                const docID = req.params.docID
                const confirmDeleteText = valid.sanitiseInput((req.body.confirmDeleteText || '').trim())

                // Pull the document that is trying to be deleted, to both check it exists but also to ensure that the user has the right permissions
                db.query('SELECT docs.*, users.username AS uploadedByUsername FROM docs LEFT JOIN users ON docs.uploadedBy = users.userID WHERE docs.docID = ?', [docID], (err, results) => {
                    if (err) {
                        console.error('Error fetching document:', err)
                        res.redirect('/docs?status=delete-error')
                        
                    } else {
                        if (!results.length) {
                            // Check that it does actually exist
                            res.sendStatus(404)
                        } else {
                            const document = results[0]

                            const canDelete = (func.hasPermissions(['ADMINISTRATOR', 'MANAGE_FILES'], userData.perms, 'OR') || func.hasPermissions([document.deletePermission], userData.perms, 'AND'))

                            if (!canDelete) {
                                res.redirect('/docs?status=forbidden')
                            } else {
                                if (confirmDeleteText.toLowerCase() !== String(document.documentName).toLowerCase()) {
                                    res.redirect('/docs?status=delete-mismatch')
                                } else {
                                    const filePath = path.resolve(__dirname, '..', 'uploads', document.filePath)

                                    db.query('DELETE FROM docs WHERE docID = ?', [document.docID], (err) => {
                                        if (err) {
                                            console.error('Error deleting document record:', err)
                                            res.redirect('/docs?status=delete-error')
                                        } else {
                                            deleteFile(filePath, () => {
                                                res.redirect('/docs?status=deleted')
                                            })
                                        }
                                    })
                                }
                            }
                        }
                    }
                })
            }
        })
    })

    router.get('/download/:docID', (req, res) => {
        func.getUserData(req.user, (success, userData) => {
            if (!success) {
                console.error('Error fetching user data:', userData)
                res.sendStatus(500)
            } else {
                const docID = req.params.docID

                db.query('SELECT docs.*, users.username AS uploadedByUsername FROM docs LEFT JOIN users ON docs.uploadedBy = users.userID WHERE docs.docID = ?', [docID], (err, results) => {
                    if (err) {
                        console.error('Error fetching document:', err)
                        res.sendStatus(500)
                    } else {
                        if (!results.length) {
                            res.sendStatus(404)
                        } else {
                            const document = results[0]

                            const hasPerms = userData.perms && func.hasPermissions(['ADMINISTRATOR', 'MANAGE_FILES'], userData.perms, 'OR') || func.hasPermissions([document.deletePermission], userData.perms, 'AND')
                            const canView = document.isPublic === 1 || hasPerms

                            if (!canView) {
                                res.sendStatus(403)
                            } else {
                                const filePath = path.resolve(__dirname, '..', 'uploads', document.filePath)

                                if (!fs.existsSync(filePath)) {
                                    res.sendStatus(404)
                                } else {
                                    res.download(filePath, document.originalFilename)
                                }
                            }
                        }
                    }
                })
            }
        })
    })

    return {
        router,
        viewsPath: path.join(__dirname, '/views'),
        staticPath: path.join(__dirname, '/public'),
        basePath: `/${moduleName}`
    }
}
